cmake_minimum_required(VERSION 3.25)
project(CubeLearner VERSION 0.2.0 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
option(BOT_CORE_TESTS_ONLY "Build SDK-independent learning tests" OFF)
if(BOT_CORE_TESTS_ONLY)
    enable_testing()
    add_subdirectory(tests)
    return()
endif()
set(GEODE_SDK "$ENV{GEODE_SDK}" CACHE PATH "Geode v5.11.2 SDK")
if(NOT GEODE_SDK)
    message(FATAL_ERROR "Set GEODE_SDK to the Geode v5.11.2 SDK directory")
endif()
file(STRINGS "${GEODE_SDK}/VERSION" BOT_SDK_VERSION LIMIT_COUNT 1)
if(NOT BOT_SDK_VERSION STREQUAL "5.11.2")
    message(FATAL_ERROR "Use SDK v5.11.2. New SDK versions require a fresh bindings and safety audit")
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This port supports Windows x64 only")
endif()
set(GEODE_TARGET_PLATFORM Win64 CACHE STRING "Geode target" FORCE)
add_subdirectory(${GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
file(GLOB_RECURSE BOT_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_library(${PROJECT_NAME} SHARED ${BOT_SOURCES})
target_include_directories(${PROJECT_NAME} PRIVATE src)
target_compile_options(${PROJECT_NAME} PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
setup_geode_mod(${PROJECT_NAME} DONT_INSTALL)
