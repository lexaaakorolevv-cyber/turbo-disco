cmake_minimum_required(VERSION 3.25)
project(RouteFinder VERSION 1.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(ROUTEFINDER_CORE_ONLY "Build portable core tests without Geometry Dash" OFF)
include(FetchContent)
FetchContent_Declare(nlohmann_json
    URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
    URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(nlohmann_json)
add_library(routefinder-core STATIC src/core.cpp src/flight.cpp)
target_include_directories(routefinder-core PUBLIC src)
target_link_libraries(routefinder-core PUBLIC nlohmann_json::nlohmann_json)
if(MSVC)
    target_compile_options(routefinder-core PRIVATE /utf-8)
endif()

if(ROUTEFINDER_CORE_ONLY)
    enable_testing()
    add_executable(routefinder-tests tests/core_tests.cpp)
    target_link_libraries(routefinder-tests PRIVATE routefinder-core)
    add_test(NAME routefinder-core COMMAND routefinder-tests)
    add_executable(routefinder-flight-tests tests/flight_tests.cpp)
    target_link_libraries(routefinder-flight-tests PRIVATE routefinder-core)
    add_test(NAME routefinder-flight COMMAND routefinder-flight-tests)
    set_tests_properties(routefinder-flight PROPERTIES TIMEOUT 90)
    return()
endif()

if(NOT DEFINED GEODE_SDK)
    set(GEODE_SDK "$ENV{GEODE_SDK}")
endif()
if(NOT EXISTS "${GEODE_SDK}/VERSION")
    message(FATAL_ERROR "Set GEODE_SDK to the official Geode v5.11.1 SDK checkout")
endif()
file(READ "${GEODE_SDK}/VERSION" SDK_VERSION)
string(STRIP "${SDK_VERSION}" SDK_VERSION)
if(NOT SDK_VERSION STREQUAL "5.11.1")
    message(FATAL_ERROR "This adapter is pinned to Geode 5.11.1 / GD 2.2081; found ${SDK_VERSION}")
endif()
include(cmake/Bindings.cmake)
add_subdirectory("${GEODE_SDK}" "${CMAKE_BINARY_DIR}/geode")
add_library(${PROJECT_NAME} SHARED src/main.cpp src/Runtime.cpp src/Menu.cpp)
target_include_directories(${PROJECT_NAME} PRIVATE src)
target_link_libraries(${PROJECT_NAME} PRIVATE routefinder-core)
if(MSVC)
    target_compile_options(${PROJECT_NAME} PRIVATE /utf-8)
endif()
setup_geode_mod(${PROJECT_NAME} DONT_INSTALL LINK_TYPE PRIVATE)
