cmake_minimum_required(VERSION 3.25)
project(RouteFinder VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(ROUTEFINDER_CORE_ONLY "Build portable engine and controller checks" OFF)
include(FetchContent)
FetchContent_Declare(nlohmann_json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(nlohmann_json)
find_package(Threads REQUIRED)
add_library(routefinder2 STATIC src/types.cpp src/search.cpp src/brain.cpp src/engine.cpp src/storage.cpp src/physics/analysis.cpp src/physics/geometry.cpp src/physics/intelligence.cpp src/physics/planner.cpp src/physics/dual.cpp src/physics/flight.cpp src/physics/ground.cpp src/physics/solo.cpp)
target_include_directories(routefinder2 PUBLIC src src/physics)
target_link_libraries(routefinder2 PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
if(MSVC)
 target_compile_options(routefinder2 PUBLIC /utf-8)
endif()
if(ROUTEFINDER_CORE_ONLY)
 enable_testing()
 foreach(name engine search persistence analysis)
  add_executable(routefinder2-${name} tests/${name}_tests.cpp)
  target_link_libraries(routefinder2-${name} PRIVATE routefinder2)
  add_test(NAME routefinder2-${name} COMMAND routefinder2-${name})
  set_tests_properties(routefinder2-${name} PROPERTIES TIMEOUT 120)
 endforeach()
 add_executable(routefinder2-modes tests/mode_tests.cpp)
 target_link_libraries(routefinder2-modes PRIVATE routefinder2)
 add_test(NAME routefinder2-modes COMMAND routefinder2-modes)
 set_tests_properties(routefinder2-modes PROPERTIES TIMEOUT 180)
 return()
endif()
if(NOT DEFINED GEODE_SDK)
 set(GEODE_SDK "$ENV{GEODE_SDK}")
endif()
if(NOT EXISTS "${GEODE_SDK}/VERSION")
 message(FATAL_ERROR "Set GEODE_SDK to official Geode 5.11.1 SDK")
endif()
file(READ "${GEODE_SDK}/VERSION" SDK_VERSION)
string(STRIP "${SDK_VERSION}" SDK_VERSION)
if(NOT SDK_VERSION STREQUAL "5.11.1")
 message(FATAL_ERROR "Requires Geode 5.11.1 / Geometry Dash 2.2081")
endif()
include(cmake/Bindings.cmake)
add_subdirectory("${GEODE_SDK}" "${CMAKE_BINARY_DIR}/geode")
add_library(${PROJECT_NAME} SHARED src/hooks.cpp src/native.cpp src/scene.cpp src/ui.cpp)
target_link_libraries(${PROJECT_NAME} PRIVATE routefinder2)
if(MSVC)
 target_compile_options(${PROJECT_NAME} PRIVATE /utf-8)
endif()
setup_geode_mod(${PROJECT_NAME} DONT_INSTALL LINK_TYPE PRIVATE)
