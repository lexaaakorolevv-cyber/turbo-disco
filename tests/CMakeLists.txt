cmake_minimum_required(VERSION 3.20)
project(cube_bot_core_tests LANGUAGES CXX)

# Standalone host tests, usable in both the SDK-free draft and final mod tree.
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../src/bot/Learner.cpp")
    set(BOT_CORE_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../src/bot/Learner.cpp")
    set(BOT_CORE_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../src")
else()
    set(BOT_CORE_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../src/Learner.cpp")
    set(BOT_CORE_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../include")
endif()
add_executable(cube_bot_core_tests core_tests.cpp "${BOT_CORE_SOURCE}")
target_include_directories(cube_bot_core_tests PRIVATE "${BOT_CORE_INCLUDE}")
target_compile_features(cube_bot_core_tests PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(cube_bot_core_tests PRIVATE /W4 /UNDEBUG)
else()
    target_compile_options(cube_bot_core_tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
endif()
option(BOT_TEST_SANITIZERS "Enable host ASan and UBSan" OFF)
if(BOT_TEST_SANITIZERS AND NOT MSVC)
    target_compile_options(cube_bot_core_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cube_bot_core_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME cube_bot_core_tests COMMAND cube_bot_core_tests)
add_executable(bot_file_commit_tests file_commit_tests.cpp ../src/FileCommit.cpp)
target_include_directories(bot_file_commit_tests PRIVATE ../src)
target_compile_features(bot_file_commit_tests PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(bot_file_commit_tests PRIVATE /W4 /UNDEBUG)
else()
    target_compile_options(bot_file_commit_tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
endif()
add_test(NAME bot_file_commit_tests COMMAND bot_file_commit_tests)
